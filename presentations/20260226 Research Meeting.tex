% SVD Sweep Overview
% ============================================================================
\begin{frame}{SVD Dimension Sweep}
\framesubtitle{Experiment design}

\textbf{Goal:} Find the optimal SVD dimensionality for MiniLM embeddings (384-d $\to$ $k$-d).

\medskip
\textbf{Setup:}
\begin{itemize}
    \item Courses: Sys Think and ML; backtest Dec 1--31, 2025.
    \item $k \in \{10, 20, 50, 100, \textit{full}\}$; \textit{full} = 384 (no SVD).
    \item SVD fitted daily on historical data only (no leakage).
\end{itemize}

\end{frame}

% ============================================================================
% Old Eval vs Oracle Eval
% ============================================================================
\begin{frame}{Old Eval vs.\ Oracle Eval}
\framesubtitle{Evaluation model used to score backtest clicks}

\textbf{Oracle model:} CV over $k \in \{10, 20, 50, 100, 384\}$; selects $k{=}384$ for both courses.

\medskip
\begin{itemize}
    \item This model evaluates the number of clicks each cost allocation produces in the backtest.
    \item Old eval: single XGBoost with fixed $k{=}50$ SVD, trained on all data.
    \item Oracle CV MSE: 1.73 (Sys Think), 2.62 (ML) --- lower than old eval (1.81, 2.81).
    \item Old exp results were overstated due to SVD data leakage and worse evaluation model.
\end{itemize}

\end{frame}

\begin{frame}{Policy Model Over-Prediction}
\framesubtitle{Gurobi-predicted clicks vs.\ Oracle-evaluated clicks}

\begin{table}[htbp]
\centering
\begin{tabular}{lccc}
\toprule
Course & Policy Clicks & Oracle Clicks & Over-prediction \\
\midrule
ML        & 8,839 & 3,202 & $2.76\times$ \\
Sys Think & 9,850 & 6,099 & $1.62\times$ \\
\bottomrule
\end{tabular}
\end{table}

\medskip
\begin{itemize}
    \item Policy clicks = $\sum_i (\hat{f}_i^{\text{policy}}(\mathbf{x}_i^*) - \hat{f}_i^{\text{policy}}(\mathbf{0}))$.
    \item Oracle clicks = $\sum_i (\hat{f}_i^{\text{oracle}}(\mathbf{x}_i^*) - \hat{f}_i^{\text{oracle}}(\mathbf{0}))$.
    \item \textbf{How can we improve?} Identify where over-prediction is high and constrain the optimizer.
\end{itemize}

\end{frame}

% ============================================================================
% Takeaways
% ============================================================================
\begin{frame}{Conclusions}

\begin{itemize}
    \item \textbf{Full embeddings (no SVD)} perform best for both courses.
    \item Policy model over-predicts clicks by $1.6$--$2.8\times$, driven by low-spend keywords.
    \item \textbf{Min-spend \$1 floor} eliminates the worst over-prediction:
    \begin{itemize}
        \item Sys Think: purchases \textbf{+42.5\%}, clicks \textbf{+77.1\%}.
        \item ML: purchases \textbf{+67.3\%}, clicks \textbf{+9.5\%}.
    \end{itemize}
    \item \textbf{Gurobi} solves the model ($>$1M variables) in $\sim$10 min; SCIP did not finish in $>$10 hours.
\end{itemize}

\end{frame}
